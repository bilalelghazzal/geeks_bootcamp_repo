SELECT film_id,
       title,
       description,
       release_year,
       rental_rate
FROM film
ORDER BY rental_rate ASC,
         film_id ASC;   -- tie-breaker for consistent ordering

       a.phone
FROM customer c
JOIN address a ON c.address_id = a.address_id
WHERE a.district = 'Texas';

12
SELECT 
    c.first_name, 
    c.last_name, 
    p.amount, 
    p.payment_date 
FROM customer c
JOIN payment p ON c.customer_id = p.customer_id 
ORDER BY c.customer_id;

13
SELECT f.*
FROM film f
WHERE NOT EXISTS (
    SELECT 1
    FROM inventory i
    WHERE i.film_id = f.film_id
);

14
SELECT 
    city.city, 
    country.country 
FROM city
JOIN country ON city.country_id = country.country_id 
ORDER BY country.country, city.city;

15
SELECT 
    c.customer_id, 
    c.first_name, 
    c.last_name, 
    p.amount, 
    p.payment_date, 
    p.staff_id
FROM customer c
JOIN payment p ON c.customer_id = p.customer_id 
ORDER BY p.staff_id;
