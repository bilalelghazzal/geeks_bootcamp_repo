-- Create the database
CREATE DATABASE bootcamp;

-- Create the students table
CREATE TABLE students (
    id SERIAL PRIMARY KEY AUTO_IN,
    last_name VARCHAR(50) NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    birth_date DATE NOT NULL
);
